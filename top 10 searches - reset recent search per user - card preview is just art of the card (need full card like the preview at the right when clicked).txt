        async function executeTextImportRoutine() {
            const rawName = document.getElementById('importDeckNameInput').value.trim();
            const rawText = document.getElementById('importRawTextarea').value;
            const processBtn = document.getElementById('runImportBtn');
            
            if (!rawName) { alert("Please allocate a target name to your new deck build profile."); return; }
            if (!rawText.trim()) { alert("The copy-paste field cannot be blank."); return; }

            processBtn.disabled = true;
            processBtn.textContent = "Processing Async Queue...";

            const lines = rawText.split('\n');
            
            let importedMainboard = {};
            let importedSideboard = {};
            let detectedCommanderCardObj = null;
            let currentBlockAreaMode = 'main'; 

            for (let line of lines) {
                let cleanLine = line.trim();
                if (!cleanLine) continue;

                let lowerLine = cleanLine.toLowerCase();
                if (lowerLine.startsWith('#') || lowerLine.startsWith('//')) {
                    if (lowerLine.includes('sideboard') || lowerLine.includes('side')) {
                        currentBlockAreaMode = 'side';
                    } else if (lowerLine.includes('commander') || lowerLine.includes('cmdr')) {
                        currentBlockAreaMode = 'commander';
                    } else {
                        currentBlockAreaMode = 'main';
                    }
                    continue;
                }

                const matchResult = cleanLine.match(/^(\d+)\s+([^#\*\(\[]+)/);
                if (!matchResult) continue;

                let quantity = parseInt(matchResult[1], 10) || 1;
                let candidateName = matchResult[2].trim();

                if (candidateName.endsWith('x')) candidateName = candidateName.slice(0, -1).trim();

                let allocationFlag = currentBlockAreaMode;
                let lowerFullLine = cleanLine.toLowerCase();

                if (lowerFullLine.includes('*sb*') || lowerFullLine.includes('sideboard') || lowerFullLine.includes('(sb)')) {
                    allocationFlag = 'side';
                } else if (lowerFullLine.includes('*cmdr*') || lowerFullLine.includes('commander') || lowerFullLine.includes('(cmdr)')) {
                    allocationFlag = 'commander';
                }

                try {
                    const fetchResponse = await fetch(`https://api.scryfall.com/cards/named?exact=${encodeURIComponent(candidateName)}`);
                    if (fetchResponse.ok) {
                        const cardData = await fetchResponse.json();
                        cacheCardMetadata(cardData);

                        if (allocationFlag === 'commander' && !detectedCommanderCardObj) {
                            detectedCommanderCardObj = cardData;
                        } else if (allocationFlag === 'side') {
                            if (!importedSideboard[cardData.id]) importedSideboard[cardData.id] = 0;
                            importedSideboard[cardData.id] += quantity;
                        } else {
                            if (!importedMainboard[cardData.id]) importedMainboard[cardData.id] = 0;
                            importedMainboard[cardData.id] += quantity;
                        }
                    }
                } catch (err) {
                    console.log(`Failed resolution pass for card entry: ${candidateName}`, err);
                }
            }

            const deckId = 'deck_' + Date.now();
            let bannerArtUrl = '';
            let commanderId = null;
            let colorIdentity = [];

            if (detectedCommanderCardObj) {
                commanderId = detectedCommanderCardObj.id;
                colorIdentity = detectedCommanderCardObj.color_identity || [];
                if (detectedCommanderCardObj.image_uris && detectedCommanderCardObj.image_uris.art_crop) {
                    bannerArtUrl = detectedCommanderCardObj.image_uris.art_crop;
                }
            } else {
                let sampleCardIds = Object.keys(importedMainboard);
                if (sampleCardIds.length > 0) {
                    let firstObj = cardMetadataCache[sampleCardIds[0]];
                    if (firstObj) colorIdentity = firstObj.color_identity || [];
                }
            }

            decklistsData[deckId] = {
                name: rawName,
                ownerId: getCurrentUserId(),
                commanderId: commanderId,
                commanderColorIdentity: colorIdentity,
                commanderBannerArt: bannerArtUrl,
                cards: importedMainboard,
                sideboard: importedSideboard
            };

            localStorage.setItem('mtgDecklistsData_v2', JSON.stringify(decklistsData));
            toggleImportModal(false);
            renderDeckSidebar();
            selectDeck(deckId);
        }

        function cacheCardMetadata(card) {
            let artCropUrl = '';
            if (card.image_uris && card.image_uris.art_crop) {
                artCropUrl = card.image_uris.art_crop;
            } else if (card.card_faces && card.card_faces[0].image_uris) {
                artCropUrl = card.card_faces[0].image_uris.art_crop;
            }

            // compute cmc if available or estimate from mana_cost
            let cmcVal = 0;
            if (typeof card.cmc === 'number') cmcVal = card.cmc;
            else if (card.mana_cost) {
                const parts = (card.mana_cost.match(/\{([^}]+)\}/g) || []);
                let sum = 0;
                parts.forEach(p => {
                    const inner = p.replace(/[{}]/g, '');
                    const n = parseInt(inner);
                    if (!isNaN(n)) sum += n; else sum += 1;
                });
                cmcVal = sum;
            }

            cardMetadataCache[card.id] = {
                id: card.id,
                name: card.name,
                set: card.set,
                set_name: card.set_name || card.set.toUpperCase(),
                type: card.type_line || 'N/A',
                mana_cost: card.mana_cost || '',
                oracle_text: card.oracle_text || '',
                cmc: cmcVal,
                layout: card.layout || 'normal',
                isTransformEligible: !!(card.card_faces && card.card_faces.length > 1) || ['transform', 'modal_dfc', 'double_faced_token', 'meld', 'flip', 'adventure'].includes(card.layout),
                prints_uri: card.prints_search_uri,
                color_identity: card.color_identity || [],
                colors: card.colors || [],
                image_uris: card.image_uris || {},
                card_faces: card.card_faces || [],
                art_crop: artCropUrl
            };
            localStorage.setItem('mtgCardMetadataCache_v2', JSON.stringify(cardMetadataCache));
        }

        function executeDirectQuantityAdjust(cardId, userId, amount) {
            if (!collectionData[cardId]) {
                // Initialize card with current user and all friends
                collectionData[cardId] = {};
                const curId = getCurrentUserId();
                collectionData[cardId][curId] = 0;
                friendsList.forEach(friend => {
                    collectionData[cardId][friend.id] = 0;
                });
            }
            
            let currentCount = collectionData[cardId][userId] || 0;
            let newCount = currentCount + amount;
            if (newCount < 0) newCount = 0;
            
            collectionData[cardId][userId] = newCount;
            
            // Check if all binders are empty
            let allEmpty = true;
            for (let userKey in collectionData[cardId]) {
                if ((collectionData[cardId][userKey] || 0) > 0) {
                    allEmpty = false;
                    break;
                }
            }
            if (allEmpty) {
                delete collectionData[cardId];
            }
            
            localStorage.setItem('mtgCollectionData_v2', JSON.stringify(collectionData));
            renderCollectionTable();
        }

        function renderCollectionTable() {
            const container = document.getElementById('collectionTableBody');
            if (!container) return;
            container.innerHTML = '';

            const cardIdsList = Object.keys(collectionData);
            let itemsWithMeta = cardIdsList.map(id => {
                return { id: id, meta: cardMetadataCache[id] || { name: 'Unknown Card', set: '???', type: 'N/A', mana_cost: '', color_identity: [] } };
            });
            itemsWithMeta.sort((a, b) => a.meta.name.localeCompare(b.meta.name));

            // Only include cards that are present in the current user's binder or in connected friends' binders
            const viewerId = getCurrentUserId();
            itemsWithMeta = itemsWithMeta.filter(item => {
                const counts = collectionData[item.id] || {};
                if ((counts[viewerId] || 0) > 0) return true;
                // also include if any connected friend actually has copies
                for (let i = 0; i < friendsList.length; i++) {
                    const f = friendsList[i];
                    if ((counts[f.id] || 0) > 0) return true;
                }
                return false;
            });

            if (currentColorFilters.length > 0) {
                itemsWithMeta = itemsWithMeta.filter(item => itemMatchesColorFilter(item.meta));
            }
            if (currentTypeFilter !== 'all') {
                itemsWithMeta = itemsWithMeta.filter(item => getCollectionCategory(item.meta.type) === currentTypeFilter);
            }

            const categories = {};
            itemsWithMeta.forEach(item => {
                const category = getCollectionCategory(item.meta.type);
                if (!categories[category]) categories[category] = [];
                categories[category].push(item);
            });

            const categoryOrder = ['Commander / Planeswalker', 'Creatures', 'Artifacts', 'Enchantments', 'Instants', 'Sorceries', 'Planeswalkers', 'Lands', 'Other'];

            categoryOrder.forEach(category => {
                const items = categories[category];
                if (!items || items.length === 0) return;
                container.appendChild(renderCollectionCategoryBlock(category, items));
                renderedAny = true;
            });

            Object.keys(categories).sort().forEach(category => {
                if (categoryOrder.includes(category)) return;
                container.appendChild(renderCollectionCategoryBlock(category, categories[category]));
                renderedAny = true;
            });

            if (!renderedAny) {
                container.innerHTML = '<div class="collection-empty-state">No cards match the selected filters or your binder is currently empty. Use the search lookup to add cards to your inventory.</div>';
                clearPreviews();
            }
            renderTradeNotifications();
        }

        function itemMatchesColorFilter(meta) {
            const identity = (meta.color_identity || []).map(c => c.toLowerCase());
            if (currentColorFilters.length === 0) return true;
            const selectedColors = currentColorFilters.filter(color => color !== 'colorless');
            const wantsColorless = currentColorFilters.includes('colorless');

            if (currentColorFilterMode === 'exact') {
                if (wantsColorless) {
                    return identity.length === 0 && selectedColors.length === 0;
                }
                return identity.length === selectedColors.length && selectedColors.every(color => identity.includes(color));
            }

            const matchesColor = identity.some(color => selectedColors.includes(color));
            const matchesColorless = wantsColorless && identity.length === 0;
            return matchesColor || matchesColorless;
        }

        function getCollectionCategory(typeLine) {
            const text = (typeLine || '').toLowerCase();
            if (text.includes('commander') || text.includes('planeswalker')) return 'Commander / Planeswalker';
            if (text.includes('creature')) return 'Creatures';
            if (text.includes('artifact') && !text.includes('creature')) return 'Artifacts';
            if (text.includes('enchantment')) return 'Enchantments';
            if (text.includes('instant')) return 'Instants';
            if (text.includes('sorcery')) return 'Sorceries';
            if (text.includes('land')) return 'Lands';
            return 'Other';
        }

        function sendTradeRequest() {
            const cardId = document.getElementById('commentCardId').value;
            const ownerId = document.getElementById('commentModalOwnerId').value;
            const type = document.getElementById('requestTypeSelect').value;
            const textarea = document.getElementById('commentInputField');
            const message = textarea.value.trim();
            if (!message) {
                alert('Please include a message with your request.');
                return;
            }
            if (ownerId === currentUser.id) {
                alert('You cannot request your own card from yourself.');
                return;
            }
            if (!tradeRequests[cardId]) tradeRequests[cardId] = [];

            const requestId = 'req_' + Date.now() + '_' + Math.random().toString(36).slice(2, 6);
            const request = {
                requestId: requestId,
                cardId: cardId,
                ownerId: ownerId,
                ownerName: getOwnerDisplayName(ownerId),
                requesterId: currentUser.id,
                requesterName: currentUser.username || 'Guest',
                type: type,
                message: message,
                status: 'pending',
                counterMessage: '',
                createdAt: new Date().toISOString(),
                updatedAt: new Date().toISOString(),
                history: [
                    { when: new Date().toISOString(), actor: currentUser.username || 'Guest', note: `Request created to ${type}.` }
                ]
            };

            tradeRequests[cardId].push(request);
            localStorage.setItem('mtgTradeRequests_v2', JSON.stringify(tradeRequests));
            textarea.value = '';
            renderRequestList(cardId, ownerId, false);
            renderTradeNotifications();
            alert('Your request has been sent. The card owner can now accept, decline, or counter the request.');
        }

        function respondToTradeRequest(cardId, requestId, action) {
            const requests = tradeRequests[cardId] || [];
            const request = requests.find(r => r.requestId === requestId);
            if (!request) return;
            if (action === 'accept') {
                request.status = 'accepted';
                request.updatedAt = new Date().toISOString();
                request.history.push({ when: request.updatedAt, actor: currentUser.username || 'Owner', note: `${getOwnerDisplayName(request.ownerId)} accepted the request.` });
            } else if (action === 'decline') {
                request.status = 'declined';
                request.updatedAt = new Date().toISOString();
                request.history.push({ when: request.updatedAt, actor: currentUser.username || 'Owner', note: `${getOwnerDisplayName(request.ownerId)} declined the request.` });
            } else if (action === 'counter') {
                const counterNote = prompt('Enter your counter-offer note:');
                if (!counterNote) return;
                request.status = 'countered';
                request.counterMessage = counterNote;
                request.updatedAt = new Date().toISOString();
                request.history.push({ when: request.updatedAt, actor: currentUser.username || 'Owner', note: `${getOwnerDisplayName(request.ownerId)} countered: ${counterNote}` });
            }
            localStorage.setItem('mtgTradeRequests_v2', JSON.stringify(tradeRequests));
            renderRequestList(cardId, request.ownerId, true);
            renderCollectionTable();
        }

        function completeTrade(cardId, requestId) {
            const requests = tradeRequests[cardId] || [];
            const request = requests.find(r => r.requestId === requestId);
            if (!request) return;
            if (request.requesterId !== currentUser.id) {
                alert('Only the requester can complete this trade.');
                return;
            }
            if (request.status !== 'accepted') {
                alert('Can only complete trades that have been accepted.');
                return;
            }

            if (!collectionData[cardId]) collectionData[cardId] = {};
            if (request.ownerId !== currentUser.id) {
                const ownerQty = collectionData[cardId][request.ownerId] || 0;
                if (ownerQty < 1) {
                    alert('The owner no longer has this card available.');
                    return;
                }
                collectionData[cardId][request.ownerId] -= 1;
            }
            const curId = getCurrentUserId();
            collectionData[cardId][curId] = (collectionData[cardId][curId] || 0) + 1;

            request.status = 'completed';
            request.updatedAt = new Date().toISOString();
            request.completedAt = request.updatedAt;
            request.history.push({ when: request.updatedAt, actor: currentUser.username || 'Requester', note: `Trade completed and card moved to ${getOwnerDisplayName(request.requesterId)} binder.` });

            localStorage.setItem('mtgCollectionData_v2', JSON.stringify(collectionData));
            localStorage.setItem('mtgTradeRequests_v2', JSON.stringify(tradeRequests));
            renderCollectionTable();
            renderTradeNotifications();
            alert('Trade completed and card added to your binder.');
        }

        function renderTradeNotifications() {
            const panel = document.getElementById('tradeNotificationsPanel');
            if (!panel) return;
            const notifications = [];
            Object.keys(tradeRequests).forEach(cardId => {
                (tradeRequests[cardId] || []).forEach(request => {
                    if (request.requesterId === currentUser.id && request.status === 'accepted') {
                        notifications.push({ cardId, request });
                    }
                });
            });

            if (notifications.length === 0) {
                panel.innerHTML = '';
                return;
            }

            panel.innerHTML = `<div style="display:flex; flex-direction:column; gap:10px; padding: 14px; background:#151515; border:1px solid #333; border-radius: 10px;">
                <div style="color:#ffcc66; font-weight:bold;">Pending Trade Notifications</div>
                ${notifications.map(note => {
                    const meta = cardMetadataCache[note.cardId] || { name: 'Unknown Card' };
                    return `<div style="display:flex; align-items:center; justify-content:space-between; gap:10px; padding:10px; background:#1d1d1d; border:1px solid #2a2a2a; border-radius:8px;">

                    </div>`;
                }).join('')}
            </div>`;
        }

        function escapeHtml(text) {
            const map = { '&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;', "'": '&#039;' };
            return text.replace(/[&<>"']/g, function(m) { return map[m]; });
        }

        async function addCardToActiveDeckDirect(cardName, targetArea) {
            if(!currentActiveDeckId) return;
            targetArea = targetArea || currentDeckSearchTarget || 'main';
            try {
                const response = await fetch(`https://api.scryfall.com/cards/named?exact=${encodeURIComponent(cardName.trim())}`);
                if(response.ok) {
                    const data = await response.json();
                    cacheCardMetadata(data);

                    const targetDeck = decklistsData[currentActiveDeckId];
                    if(!targetDeck.sideboard) targetDeck.sideboard = {};
                    if(!targetDeck.cards) targetDeck.cards = {};

                    if (targetArea === 'side') {
                        if(!targetDeck.sideboard[data.id]) targetDeck.sideboard[data.id] = 0;
                        targetDeck.sideboard[data.id]++;
                    } else {
                        if(!targetDeck.cards[data.id]) targetDeck.cards[data.id] = 0;
                        targetDeck.cards[data.id]++;
                    }
                    
                    localStorage.setItem('mtgDecklistsData_v2', JSON.stringify(decklistsData));
                    document.getElementById('deckCardSearchInput').value = '';
                    renderActiveDeckComposition();
                }
            } catch(e){ console.log(e); }
        }

        function adjustCardInDeck(cardId, amount, areaTarget) {
            if(!currentActiveDeckId) return;
            const targetDeck = decklistsData[currentActiveDeckId];
            
            if (areaTarget === 'side') {
                if (targetDeck.sideboard && targetDeck.sideboard[cardId] !== undefined) {
                    targetDeck.sideboard[cardId] += amount;
                    if(targetDeck.sideboard[cardId] <= 0) delete targetDeck.sideboard[cardId];
                }
            } else {
                if (targetDeck.cards && targetDeck.cards[cardId] !== undefined) {
                    targetDeck.cards[cardId] += amount;
                    if(targetDeck.cards[cardId] <= 0) delete targetDeck.cards[cardId];
                }
            }
            
            localStorage.setItem('mtgDecklistsData_v2', JSON.stringify(decklistsData));
            renderActiveDeckComposition();
        }

        /* --- DECK EDITING CONTROL OPERATIONS (3-DOTS ARCHITECTURE) --- */
        function changeCardZoneDestination(cardId, currentZone) {
            if(!currentActiveDeckId) return;
            const targetDeck = decklistsData[currentActiveDeckId];
            
            if(currentZone === 'main') {
                const count = targetDeck.cards[cardId] || 1;
                delete targetDeck.cards[cardId];
                targetDeck.sideboard[cardId] = (targetDeck.sideboard[cardId] || 0) + count;
            } else {
                const count = targetDeck.sideboard[cardId] || 1;
                delete targetDeck.sideboard[cardId];
                targetDeck.cards[cardId] = (targetDeck.cards[cardId] || 0) + count;
            }
            
            localStorage.setItem('mtgDecklistsData_v2', JSON.stringify(decklistsData));
            renderActiveDeckComposition();
        }

        function removeCardFromDeckCompletely(cardId, zone) {
            if(!currentActiveDeckId) return;
            const targetDeck = decklistsData[currentActiveDeckId];
            if (zone === 'side') delete targetDeck.sideboard[cardId];
            else delete targetDeck.cards[cardId];
            
            localStorage.setItem('mtgDecklistsData_v2', JSON.stringify(decklistsData));
            renderActiveDeckComposition();
        }

        function promoteCardToDeckCommander(cardId, sourceZone) {
            if(!currentActiveDeckId) return;
            const targetDeck = decklistsData[currentActiveDeckId];
            const meta = cardMetadataCache[cardId];
            
            if(!meta) return;

            if (sourceZone === 'side') {
                if(targetDeck.sideboard[cardId] > 1) targetDeck.sideboard[cardId]--;
                else delete targetDeck.sideboard[cardId];
            } else {
                if(targetDeck.cards[cardId] > 1) targetDeck.cards[cardId]--;
                else delete targetDeck.cards[cardId];
            }

            if (targetDeck.commanderId) {
                const oldCommId = targetDeck.commanderId;
                targetDeck.cards[oldCommId] = (targetDeck.cards[oldCommId] || 0) + 1;
            }

            targetDeck.commanderId = cardId;
            targetDeck.commanderColorIdentity = meta.color_identity || [];
            targetDeck.commanderBannerArt = meta.art_crop || '';

            document.getElementById('activeDeckIdentityText').textContent = `Commander Identity: ${meta.color_identity.length === 0 ? 'Colorless (C)' : meta.color_identity.join(', ')}`;
            if (meta.art_crop) {
                document.getElementById('deckBannerHeader').style.backgroundImage = `url('${meta.art_crop}')`;
            }

            localStorage.setItem('mtgDecklistsData_v2', JSON.stringify(decklistsData));
            renderActiveDeckComposition();
        }

            const activeDeck = decklistsData[currentActiveDeckId];

                "Artifacts": [],
                "Enchantments": [],
                "Lands": []
            };

            if (activeDeck.commanderId) {
                const commMeta = cardMetadataCache[activeDeck.commanderId] || { name: 'Unknown Commander', set: '???' };
                structuredCategories["Commander"].push({ id: activeDeck.commanderId, meta: commMeta, count: 1, isCommanderRow: true });
                sumMainCount++; 
            }

            const deckCardIds = Object.keys(activeDeck.cards);
            deckCardIds.forEach(id => {
                const meta = cardMetadataCache[id] || { name: 'Unknown', set: '???', type: '' };
                const typeLine = (meta.type || '').toLowerCase();
                const countVal = activeDeck.cards[id];
                sumMainCount += countVal;
                
                let assigned = false;
                if (typeLine.includes('land')) { structuredCategories["Lands"].push({ id: id, meta: meta, count: countVal }); assigned = true; }
                else if (typeLine.includes('creature')) { structuredCategories["Creatures"].push({ id: id, meta: meta, count: countVal }); assigned = true; }
                else if (typeLine.includes('instant')) { structuredCategories["Instants"].push({ id: id, meta: meta, count: countVal }); assigned = true; }
                else if (typeLine.includes('sorcery')) { structuredCategories["Sorceries"].push({ id: id, meta: meta, count: countVal }); assigned = true; }
                else if (typeLine.includes('artifact')) { structuredCategories["Artifacts"].push({ id: id, meta: meta, count: countVal }); assigned = true; }
                else if (typeLine.includes('enchantment')) { structuredCategories["Enchantments"].push({ id: id, meta: meta, count: countVal }); assigned = true; }
                
                if(!assigned) {
                    structuredCategories["Creatures"].push({ id: id, meta: meta, count: countVal });
                }
            });

            const sideCardIds = Object.keys(activeDeck.sideboard);
            sideCardIds.forEach(id => {
                const meta = cardMetadataCache[id] || { name: 'Unknown', set: '???', type: '' };
                const countVal = activeDeck.sideboard[id];
                sumSideCount += countVal;
                sideboardCategoryList.push({ id: id, meta: meta, count: countVal, isSideboardItem: true });

            if (!galleryList || galleryList.length === 0) return;
            galleryIndex = (idx + galleryList.length) % galleryList.length;
            const id = galleryList[galleryIndex];
            let meta = cardMetadataCache[id];
            if (!meta) {
                try {
                    const resp = await fetch(`https://api.scryfall.com/cards/${id}`);
                    if (resp.ok) {
                        meta = await resp.json();
                        cacheCardMetadata(meta);
                    }
                } catch(e) { console.log(e); }
            }
            const img = document.getElementById('cardGalleryImage');
            const details = document.getElementById('cardGalleryDetails');
            if (meta) {
                const imageUrl = getCardPreviewImage(meta, 0) || meta.image_uris?.normal || meta.art_crop || meta.image_uris?.art_crop || '';
                img.src = imageUrl;
                let name = meta.name || (meta.card_faces && meta.card_faces[0] && meta.card_faces[0].name) || 'Unknown';
                let mana = meta.mana_cost || '';
                let typeLine = meta.type_line || '';
                let oracle = meta.oracle_text || (meta.card_faces && meta.card_faces.map(f=>f.oracle_text).filter(Boolean).join('\n\n')) || '';
                let html = `<div style="font-weight:700;">${escapeHtml(name)}</div>`;
                if (mana || typeLine) html += `<div style="color:#aaa; margin-bottom:8px;">${escapeHtml(mana)} ${escapeHtml(typeLine)}</div>`;
                if (oracle) html += `<div style="white-space:pre-wrap; color:#ddd;">${escapeHtml(oracle)}</div>`;
                details.innerHTML = html;
            } else {
                img.src = '';
                details.textContent = 'Card data not available';
            }
            const prevBtn = document.getElementById('cardPrevBtn');
            const nextBtn = document.getElementById('cardNextBtn');
            prevBtn.onclick = () => showGalleryAtIndex(galleryIndex - 1);
            nextBtn.onclick = () => showGalleryAtIndex(galleryIndex + 1);
        }

        function removeFriend(friendId, friendName) {
            if (!confirm(`Are you sure you want to remove "${friendName}" from your network?`)) {
                return;
            }

            // Remove friend from friends list
            friendsList = friendsList.filter(f => f.id !== friendId);
            updateFriendsListStorage();

            // Clean up collection data - remove this friend's data from all cards
            Object.keys(collectionData).forEach(cardId => {
                if (collectionData[cardId][friendId]) {
                    delete collectionData[cardId][friendId];
                }
                // If card now has no holders with quantity, delete it entirely
                if (Object.keys(collectionData[cardId]).length === 0 || 
                    Object.values(collectionData[cardId]).every(v => !v || v === 0)) {
                    delete collectionData[cardId];
                }
            });
            localStorage.setItem('mtgCollectionData_v2', JSON.stringify(collectionData));

            renderFriendsGrid();
            renderCollectionTable();
        }

        function ensureUniqueUserIds() {
            // Build reverse map of id -> [usernames]
            const idMap = {};
            Object.keys(users).forEach(username => {
                const u = users[username] || {};
                const id = u.id || null;
                if (!id) return;
                if (!idMap[id]) idMap[id] = [];
                idMap[id].push(username);
            });

            let modified = false;
            Object.keys(idMap).forEach(id => {
                const names = idMap[id];
                if (names.length <= 1) return;
                // Keep the first username mapped to the existing id; give others new ids
                for (let i = 1; i < names.length; i++) {
                    const uname = names[i];
                    const newId = uname.startsWith('Guest') || users[uname].guest ? 'guest_' + Date.now() + '_' + i : 'user_' + Date.now() + '_' + i;
                    users[uname].id = newId;
                    // Initialize collectionData slots for the new id to avoid referencing the old shared id
                    Object.keys(collectionData).forEach(cardId => {
                        if (!collectionData[cardId]) collectionData[cardId] = {};
                        if (!collectionData[cardId][newId]) collectionData[cardId][newId] = 0;
                    });
                    modified = true;
                }
            });

            if (modified) {
                localStorage.setItem('mtgUsers_v2', JSON.stringify(users));
                localStorage.setItem('mtgCollectionData_v2', JSON.stringify(collectionData));
                // If currentUser was one of the changed users, update the persisted currentUser
                if (currentUser && users[currentUser.username] && users[currentUser.username].id !== currentUser.id) {
                    currentUser.id = users[currentUser.username].id;
                    localStorage.setItem('mtgCurrentUser_v2', JSON.stringify(currentUser));
                }
            }
        }

        function getCurrentUserId() {
            return (currentUser && currentUser.id) ? currentUser.id : 'guest_anonymous';
        }

        function ensureCollectionDataStructure() {
            const curId = getCurrentUserId();
            Object.keys(collectionData).forEach(cardId => {
                if (!collectionData[cardId]) collectionData[cardId] = {};
                if (!collectionData[cardId][curId]) collectionData[cardId][curId] = 0;
                friendsList.forEach(f => {
                    if (!collectionData[cardId][f.id]) collectionData[cardId][f.id] = 0;
                });
            });
            // persist any new keys
            localStorage.setItem('mtgCollectionData_v2', JSON.stringify(collectionData));
        }

        function ensureDeckOwnershipDataStructure() {

                if (!decklistsData[deckId].ownerId) {
                    decklistsData[deckId].ownerId = curId;
                    modified = true;
                }
            });

                localStorage.setItem('mtgDecklistsData_v2', JSON.stringify(decklistsData));
            }
        }

        function getCmcFromMeta(meta) {
            if (typeof meta.cmc === 'number') return Math.max(0, meta.cmc);
            if (meta.mana_cost) {
                const parts = meta.mana_cost.match(/\{([^}]+)\}/g) || [];
                return parts.reduce((sum, part) => {
                    const inner = part.replace(/[{}]/g, '');
                    const numeric = parseInt(inner);
                    return sum + (isNaN(numeric) ? 1 : numeric);
                }, 0);
            }
            if (meta.card_faces && Array.isArray(meta.card_faces)) {
                return meta.card_faces.reduce((sum, face) => {
                    const faceCost = face.mana_cost || '';
                    const parts = faceCost.match(/\{([^}]+)\}/g) || [];
                    return sum + parts.reduce((sub, part) => {
                        const inner = part.replace(/[{}]/g, '');
                        const numeric = parseInt(inner);
                        return sub + (isNaN(numeric) ? 1 : numeric);
                    }, 0);
                }, 0);
            }
            return 0;
        }

        function computeAndRenderDeckStats(deck) {
            const main = deck.cards || {};
            const colors = ['W','U','B','R','G','C'];

            const colorSymbols = {W:'{W}',U:'{U}',B:'{B}',R:'{R}',G:'{G}',C:'{C}'};
            const colorMap = {W:'#fff8d6',U:'#3b82f6',B:'#111827',R:'#ef4444',G:'#22c55e',C:'#9ca3af'};

            const avgTotals = {W:0,U:0,B:0,R:0,G:0,C:0};
            const avgCounts = {W:0,U:0,B:0,R:0,G:0,C:0};
            const bins = Array.from({length:11}, ()=> ({perm: 0, nonperm: 0}));
            const bucketCards = Array.from({length:11}, ()=> ({perm: [], nonperm: []}));
            const colorCards = {W:{perm:[],nonperm:[]},U:{perm:[],nonperm:[]},B:{perm:[],nonperm:[]},R:{perm:[],nonperm:[]},G:{perm:[],nonperm:[]},C:{perm:[],nonperm:[]}};

            Object.keys(main).forEach(cardId => {
                const qty = main[cardId] || 0;
                if (!qty) return;
                const meta = cardMetadataCache[cardId] || {cmc:0,color_identity:[],colors:[],type:'N/A',name:'Unknown',mana_cost:'',card_faces:[]};
                const cmc = Math.max(0, Math.round(getCmcFromMeta(meta)));
                const isPermanent = !(/instant|sorcery/i).test(meta.type || '');
                const isLand = /land/i.test(meta.type || '');
                if (isLand) return;
                const colorsForCard = (meta.color_identity && meta.color_identity.length) ? meta.color_identity : (meta.colors && meta.colors.length ? meta.colors : ['C']);
                const bucketIndex = Math.min(10, Math.floor(cmc));

                const entry = { id: cardId, name: meta.name, qty, cmc, type: meta.type, colors: colorsForCard };
                if (isPermanent) bucketCards[bucketIndex].perm.push(entry);
                else bucketCards[bucketIndex].nonperm.push(entry);

                colorsForCard.forEach(c => {
                    const key = colors.includes(c) ? c : 'C';
                    avgTotals[key] += cmc * qty;
                    avgCounts[key] += qty;
                    if (isPermanent) colorCards[key].perm.push(entry);
                    else colorCards[key].nonperm.push(entry);
                });
                if (isPermanent) bins[bucketIndex].perm += qty;
                else bins[bucketIndex].nonperm += qty;
            });

            const avgContainer = document.getElementById('avgCmcPerColor');
            avgContainer.innerHTML = '';
            colors.forEach(c => {
                const avg = avgCounts[c] ? (avgTotals[c] / avgCounts[c]).toFixed(2) : '-';
                const chip = document.createElement('div');

                chip.style.color = '#ddd';

                const manaIcon = document.createElement('span');
                manaIcon.style.display = 'inline-flex';
                manaIcon.style.alignItems = 'center';
                manaIcon.style.justifyContent = 'center';
                manaIcon.style.width = '18px';
                manaIcon.style.height = '18px';
                manaIcon.style.borderRadius = '50%';
                manaIcon.style.background = colorMap[c] || '#888';
                manaIcon.style.color = c === 'W' ? '#111' : '#fff';
                manaIcon.style.fontWeight = '700';
                manaIcon.style.fontSize = '12px';
                manaIcon.textContent = c === 'C' ? '◇' : c;
                chip.appendChild(manaIcon);
                const avgText = document.createElement('span');
                avgText.textContent = avg;
                chip.appendChild(avgText);
                avgContainer.appendChild(chip);
            });

            renderManaHistogram(bins, bucketCards, colorCards, colorMap, colorSymbols, colorNames);
        }

        function renderManaHistogram(bins, bucketCards, colorCards, colorMap, colorSymbols, colorNames) {
            const svg = document.getElementById('manaHistogram');
            const legend = document.getElementById('manaHistogramLegend');
            const listContainer = document.getElementById('manaHistogramList');
            svg.innerHTML = '';
            legend.innerHTML = '';
            listContainer.innerHTML = '';

            const width = 600, height = 190, padding = 30;
            const binsCount = bins.length;
            const barBaseWidth = (width - padding * 2) / binsCount;
            const barWidth = barBaseWidth * 0.7;
            const gap = barBaseWidth - barWidth;
            const permColor = '#f59e0b';
            const nonpermColor = '#9ca3af';

            let maxTotal = 0;
            for (let i = 0; i < binsCount; i++) {
                const tot = (bins[i].perm || 0) + (bins[i].nonperm || 0);
                maxTotal = Math.max(maxTotal, tot);
            }
            const scale = maxTotal > 0 ? ((height - 60) / maxTotal) : 0;

            for (let i = 0; i < binsCount; i++) {
                const x = padding + i * barBaseWidth + gap / 2;
                const bucket = bins[i];
                const group = document.createElementNS('http://www.w3.org/2000/svg','g');

                const permHeight = (bucket.perm || 0) * scale;
                if (permHeight > 0) {
                    const rect = document.createElementNS('http://www.w3.org/2000/svg','rect');
                    rect.setAttribute('x', x);
                    rect.setAttribute('y', height - 20 - permHeight);
                    rect.setAttribute('width', barWidth);
                    rect.setAttribute('height', permHeight);
                    rect.setAttribute('fill', permColor);
                    rect.setAttribute('stroke', deckStatsSelection.bucket === i && deckStatsSelection.section === 'perm' ? '#fff' : '#111');
                    rect.setAttribute('stroke-width', deckStatsSelection.bucket === i && deckStatsSelection.section === 'perm' ? '1.5' : '0.5');
                    rect.style.cursor = 'pointer';
                    rect.onclick = () => {
                        deckStatsSelection.bucket = i;
                        deckStatsSelection.section = deckStatsSelection.section === 'perm' && deckStatsSelection.bucket === i ? null : 'perm';
                        renderManaHistogram(bins, bucketCards, colorCards, colorMap, colorSymbols, colorNames);
                    };
                    group.appendChild(rect);
                }

                const nonpermHeight = (bucket.nonperm || 0) * scale;
                if (nonpermHeight > 0) {
                    const rect = document.createElementNS('http://www.w3.org/2000/svg','rect');
                    rect.setAttribute('x', x);
                    rect.setAttribute('y', height - 20 - permHeight - nonpermHeight);
                    rect.setAttribute('width', barWidth);
                    rect.setAttribute('height', nonpermHeight);
                    rect.setAttribute('fill', nonpermColor);
                    rect.setAttribute('opacity', '0.9');
                    rect.setAttribute('stroke', deckStatsSelection.bucket === i && deckStatsSelection.section === 'nonperm' ? '#fff' : '#111');
                    rect.setAttribute('stroke-width', deckStatsSelection.bucket === i && deckStatsSelection.section === 'nonperm' ? '1.5' : '0.5');
                    rect.style.cursor = 'pointer';
                    rect.onclick = () => {
                        deckStatsSelection.bucket = i;
                        deckStatsSelection.section = deckStatsSelection.section === 'nonperm' && deckStatsSelection.bucket === i ? null : 'nonperm';
                        renderManaHistogram(bins, bucketCards, colorCards, colorMap, colorSymbols, colorNames);
                    };
                    group.appendChild(rect);
                }

                const label = document.createElementNS('http://www.w3.org/2000/svg','text');
                label.setAttribute('x', x + barWidth / 2);
                label.setAttribute('y', height - 2);
                label.setAttribute('fill', '#aaa');
                label.setAttribute('font-size', '10');
                label.setAttribute('text-anchor', 'middle');
                label.textContent = i === 10 ? '10+' : String(i);
                group.appendChild(label);
                svg.appendChild(group);
            }

            const legendItems = [
                { label: 'Permanents', color: permColor, section: 'perm' },
                { label: 'Non-Permanents', color: nonpermColor, section: 'nonperm' }
            ];

            legendItems.forEach(item => {
                const chip = document.createElement('div');
                chip.style.display = 'inline-flex';
                chip.style.alignItems = 'center';
                chip.style.cursor = 'pointer';
                chip.style.padding = '6px 10px';
                chip.style.borderRadius = '8px';
                chip.style.background = deckStatsSelection.section === item.section ? '#1f2937' : '#111827';
                chip.style.border = deckStatsSelection.section === item.section ? '1px solid #fff' : '1px solid #111827';
                chip.style.color = '#ddd';
                chip.style.fontSize = '13px';
                chip.style.gap = '8px';
                const colorDot = document.createElement('span');
                colorDot.style.width = '14px';
                colorDot.style.height = '14px';
                colorDot.style.borderRadius = '50%';
                colorDot.style.background = item.color;
                chip.appendChild(colorDot);
                const label = document.createElement('span');
                label.textContent = item.label;
                chip.appendChild(label);
                chip.onclick = () => {
                    if (deckStatsSelection.section === item.section) {
                        deckStatsSelection.section = null;
                    } else {
                        deckStatsSelection.section = item.section;
                        deckStatsSelection.bucket = null;
                    }
                    renderManaHistogram(bins, bucketCards, colorCards, colorMap, colorSymbols, colorNames);
                };
                legend.appendChild(chip);
            });

            renderHistogramList(bucketCards, colorCards);
        }

        function renderHistogramList(bucketCards, colorCards) {
            const container = document.getElementById('manaHistogramList');
            container.innerHTML = '';

            const selectedBucket = deckStatsSelection.bucket;
            const selectedSection = deckStatsSelection.section;

            const title = document.createElement('div');
            title.style.marginBottom = '10px';
            title.style.color = '#bbb';
            if (selectedBucket === null && !selectedSection) {
                title.textContent = 'Click a CMC bar segment to inspect matching deck cards.';
                container.appendChild(title);
                return;
            }

            let perm = [];
            let nonperm = [];
            if (selectedBucket !== null) {
                perm = bucketCards[selectedBucket].perm.slice();
                nonperm = bucketCards[selectedBucket].nonperm.slice();
            } else if (selectedSection === 'perm' || selectedSection === 'nonperm') {
                bucketCards.forEach(bucket => {
                    if (selectedSection === 'perm') perm = perm.concat(bucket.perm);
                    else nonperm = nonperm.concat(bucket.nonperm);
                });
            }

            let detailLabel = selectedBucket !== null ? `CMC ${selectedBucket === 10 ? '10+' : selectedBucket}` : 'Selected';
            if (selectedSection) {
                detailLabel += selectedBucket !== null ? ` - ${selectedSection === 'perm' ? 'Permanents' : 'Non-Permanents'}` : ` - ${selectedSection === 'perm' ? 'Permanents' : 'Non-Permanents'}`;
            }
            title.textContent = detailLabel;
            container.appendChild(title);

            if (!selectedSection || selectedSection === 'perm') {
                const permTitle = document.createElement('div');
                permTitle.style.color = '#4ade80';
                permTitle.style.fontWeight = '700';
                permTitle.textContent = `Permanents (${perm.reduce((sum, card) => sum + card.qty, 0)})`;
                container.appendChild(permTitle);
                if (perm.length === 0) {
                    const empty = document.createElement('div');
                    empty.style.color = '#666';
                    empty.style.marginBottom = '8px';
                    empty.textContent = 'No permanent cards match this selection.';
                    container.appendChild(empty);
                } else {
                    perm.forEach(it => {
                        const row = document.createElement('div');
                        row.style.color = '#ddd';
                        row.style.fontSize = '13px';
                        row.style.marginBottom = '4px';
                        const nameSpan = document.createElement('span');
                        nameSpan.className = 'clickable-card-name';
                        nameSpan.textContent = `${it.qty}x [${it.cmc}] ${it.name}`;
                        nameSpan.style.cursor = 'pointer';
                        nameSpan.onclick = () => triggerRightSideCardLookupById(it.id);
                        nameSpan.onmouseenter = () => {
                            const meta = cardMetadataCache[it.id];
                            if (meta) {
                                showHoverTooltip(meta);
                            } else {
                                fetch(`https://api.scryfall.com/cards/${it.id}`).then(r => r.json()).then(card => {
                                    showHoverTooltip(card);
                                }).catch(() => {});
                            }
                        };
                        nameSpan.onmouseleave = hideHoverTooltip;
                        row.appendChild(nameSpan);
                        container.appendChild(row);
                    });
                }
            }

            if (!selectedSection || selectedSection === 'nonperm') {
                const nonTitle = document.createElement('div');
                nonTitle.style.color = '#60a5fa';
                nonTitle.style.fontWeight = '700';
                nonTitle.style.marginTop = '10px';
                nonTitle.textContent = `Non-Permanents (${nonperm.reduce((sum, card) => sum + card.qty, 0)})`;
                container.appendChild(nonTitle);
                if (nonperm.length === 0) {
                    const empty = document.createElement('div');
                    empty.style.color = '#666';
                    empty.textContent = 'No non-permanent cards match this selection.';
                    container.appendChild(empty);
                } else {
                    nonperm.forEach(it => {
                        const row = document.createElement('div');
                        row.style.color = '#ddd';
                        row.style.fontSize = '13px';
                        row.style.marginBottom = '4px';
                        const nameSpan = document.createElement('span');
                        nameSpan.className = 'clickable-card-name';
                        nameSpan.textContent = `${it.qty}x [${it.cmc}] ${it.name}`;
                        nameSpan.style.cursor = 'pointer';
                        nameSpan.onclick = () => triggerRightSideCardLookupById(it.id);
                        nameSpan.onmouseenter = () => {
                            const meta = cardMetadataCache[it.id];
                            if (meta) {
                                showHoverTooltip(meta);
                            } else {
                                fetch(`https://api.scryfall.com/cards/${it.id}`).then(r => r.json()).then(card => {
                                    showHoverTooltip(card);
                                }).catch(() => {});
                            }
                        };
                        nameSpan.onmouseleave = hideHoverTooltip;
                        row.appendChild(nameSpan);
                        container.appendChild(row);
                    });
                }
            }
        }
    </script>
